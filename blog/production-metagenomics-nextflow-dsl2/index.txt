workflow VIRAL_SURVEILLANCE_PIPELINE {
    take:
        ch_samplesheet // channel: [ val(meta), path(reads) ]
        ch_host_index  // path: host genome index
        ch_kraken_db   // path: viral kraken2 db

    main:
        // Step 1: Splice-aware host read filtering
        HOST_DEPLETION(ch_samplesheet, ch_host_index)

        // Step 2: High-speed k-mer taxonomic profiling
        KRAKEN2_VIRAL(HOST_DEPLETION.out.clean_reads, ch_kraken_db)

        // Step 3: De novo viral contig assembly
        SPADES_ASSEMBLY(HOST_DEPLETION.out.clean_reads)

        // Step 4: MultiQC and interactive Krona charts
        QUALITY_REPORT(
            HOST_DEPLETION.out.stats,
            KRAKEN2_VIRAL.out.reports,
            SPADES_ASSEMBLY.out.contigs
        )

    emit:
        contigs = SPADES_ASSEMBLY.out.contigs
        reports = QUALITY_REPORT.out.html_report
